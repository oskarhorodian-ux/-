local function recalculateOrders()
    orderCounter = 0
    local sorted = {}
    for _, pt in ipairs(savedPoints) do
        if pt.Order and pt.Order > 0 then table.insert(sorted, pt) end
    end
    table.sort(sorted, function(a,b) return a.Order < b.Order end)
    for i, pt in ipairs(sorted) do
        pt.Order = i
        if pt.NumBox then pt.NumBox.Text = tostring(i) end
        orderCounter = i
    end
end

local function moveToDest(hrp, dest, abortFn)
    if isTeleportMode then
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.CFrame = CFrame.new(dest)
        return
    end
    local wasAnc = hrp.Anchored
    if not dropOnPoint then hrp.Anchored = true end
    while hrp and hrp.Parent and not _G.ScriptDestroyed do
        if abortFn and abortFn() then break end
        local diff = dest - hrp.Position
        local dist = diff.Magnitude
        if dist <= 1.8 then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.CFrame = CFrame.new(dest)
            break
        end
        local dt = RunService.Heartbeat:Wait()
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.CFrame = CFrame.new(hrp.Position + diff.Unit * math.min(dist, ziplineSpeedValue * dt))
    end
    if hrp and hrp.Parent and not invisActive then hrp.Anchored = wasAnc end
end

local ziplineActive = false
runZiplineBtn.MouseButton1Click:Connect(function()
    if ziplineActive then ziplineActive = false return end
    local char = Player.Character
    local hrp = (invisActive and ghostSimulationCube) or (godmodeEnabled and godGroundCube) or (char and char:FindFirstChild("HumanoidRootPart"))
    if not hrp then return end

    ziplineActive = true
    runZiplineBtn.BackgroundColor3 = Color3.fromRGB(120, 100, 20)
    runZiplineBtn.Text = "🛑 Стоп"

    task.spawn(function()
        repeat
            local ordered = {}
            for _, pt in ipairs(savedPoints) do
                if pt.Order and pt.Order > 0 then table.insert(ordered, pt) end
            end
            table.sort(ordered, function(a, b) return a.Order < b.Order end)
            if #ordered == 0 then break end

            for _, pt in ipairs(ordered) do
                if not ziplineActive or _G.ScriptDestroyed then break end
                local r = (invisActive and ghostSimulationCube) or (godmodeEnabled and godGroundCube) or (Player.Character and Player.Character:FindFirstChild("HumanoidRootPart"))
                if not r then break end
                moveToDest(r, pt.Pos + Vector3.new(0, 3, 0), function() return not ziplineActive end)
                if ziplineDelayValue > 0 and ziplineActive then task.wait(ziplineDelayValue) end
            end
            task.wait(0.05)
        until not isZiplineLoop or not ziplineActive or _G.ScriptDestroyed

        ziplineActive = false
        runZiplineBtn.BackgroundColor3 = Color3.fromRGB(220, 180, 40)
        runZiplineBtn.Text = "🚀 Старт"
    end)
end)

local function bindDeathTracker(char)
    if not char then return end
    local hum = char:WaitForChild("Humanoid", 5)
    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if hum and hrp then
        hum.Died:Connect(function() if hrp and hrp.Parent then lastDeathPos = hrp.Position end end)
    end
end
if Player.Character then task.spawn(bindDeathTracker, Player.Character) end

Player.CharacterAdded:Connect(function(newChar)
    task.spawn(bindDeathTracker, newChar)
    if not deathZiplineEnabled or _G.ScriptPaused or _G.ScriptDestroyed then return end
    local newHrp = newChar:WaitForChild("HumanoidRootPart", 5)
    if not newHrp then return end
    task.wait(0.45)

    local ordered = {}
    for _, pt in ipairs(savedPoints) do
        if pt.Order and pt.Order > 0 then table.insert(ordered, pt) end
    end
    table.sort(ordered, function(a, b) return a.Order < b.Order end)
    if #ordered == 0 then return end

    local backPos = lastDeathPos or newHrp.Position
    for _, pt in ipairs(ordered) do
        if not deathZiplineEnabled or _G.ScriptDestroyed or not newHrp.Parent then break end
        moveToDest(newHrp, pt.Pos + Vector3.new(0, 3, 0), function() return not deathZiplineEnabled end)
        if ziplineDelayValue > 0 then task.wait(ziplineDelayValue) end
    end
    if deathZiplineEnabled and newHrp.Parent and backPos then
        moveToDest(newHrp, backPos + Vector3.new(0, 3, 0), function() return not deathZiplineEnabled end)
    end
end)
