        camSpeedBox.PlaceholderText = "Число..."
        camSpeedBox.TextColor3 = Color3.fromRGB(100, 220, 255)

        camSpeedBox.ClearTextOnFocus = true
        camSpeedBox.ZIndex = 55
        Instance.new("UICorner", camSpeedBox).CornerRadius = UDim.new(0, 5)

        camSpeedBox.FocusLost:Connect(function()
            local num = tonumber(camSpeedBox.Text)
            if num and num > 0 then flySpeedValue = num end
            camSpeedBox.Text = "Скорость: " .. flySpeedValue
        end)

        local wasAnchored = hrp and hrp.Anchored or false
        if hrp and not phantomRunning then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.Anchored = true
        end

        local camPos = cam.CFrame.Position
        local rx, ry, _ = cam.CFrame:ToEulerAnglesYXZ()
        local pitch = rx
        local yaw = ry
        cam.CameraType = Enum.CameraType.Scriptable

        local UIS = game:GetService("UserInputService")
        local lookConn = UIS.InputChanged:Connect(function(input, gp)
            if not freecamEnabled or gp then return end
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
                if input.Position.X < 200 and input.Position.Y > (cam.ViewportSize.Y - 200) then return end
                yaw = yaw - (input.Delta.X * 0.0045)
                pitch = math.clamp(pitch - (input.Delta.Y * 0.0045), -1.45, 1.45)
            end
        end)

        task.spawn(function()
            local lastT = tick()
            while freecamEnabled and not _G.ScriptDestroyed do
                local dt = tick() - lastT
                lastT = tick()

                local rot = CFrame.fromEulerAnglesYXZ(pitch, yaw, 0)
                if not _G.ScriptPaused then
                    local moveVec = Vector3.new(0, 0, 0)
                    if keysPressed.w or UIS:IsKeyDown(Enum.KeyCode.W) then moveVec = moveVec + rot.LookVector end
                    if keysPressed.s or UIS:IsKeyDown(Enum.KeyCode.S) then moveVec = moveVec - rot.LookVector end
                    if keysPressed.a or UIS:IsKeyDown(Enum.KeyCode.A) then moveVec = moveVec - rot.RightVector end
                    if keysPressed.d or UIS:IsKeyDown(Enum.KeyCode.D) then moveVec = moveVec + rot.RightVector end

                    if moveVec.Magnitude > 0 then
                        camPos = camPos + moveVec.Unit * (flySpeedValue * dt)
                    end
                end
                cam.CFrame = CFrame.new(camPos) * rot
                RunService.RenderStepped:Wait()
            end

            if lookConn then lookConn:Disconnect() end
            keysPressed = {w = false, a = false, s = false, d = false}
            destroyAllWasdFrames()
            cam.CameraType = Enum.CameraType.Custom
            if hrp and hrp.Parent and not phantomRunning and not airFreezeEnabled then
                hrp.Anchored = wasAnchored
            end
            if phantomRunning and ghostVisualClone then
                cam.CameraSubject = ghostVisualClone:FindFirstChild("Head") or ghostSimulationCube
            elseif hum and hum.Parent then
                cam.CameraSubject = hum
            end
        end)
    else
        keysPressed = {w = false, a = false, s = false, d = false}

        cam.CameraType = Enum.CameraType.Custom
        if hrp and hrp.Parent and not phantomRunning and not airFreezeEnabled then
            hrp.Anchored = false
        end
        if phantomRunning and ghostVisualClone then
            cam.CameraSubject = ghostVisualClone:FindFirstChild("Head") or ghostSimulationCube
        elseif hum and hum.Parent then
            cam.CameraSubject = hum
        end
    end

local targetPlayerInput = ""
local followBehindEnabled = false
local spectatingPlayer = nil

local function findExactOrPrefixPlayer(query)
    if not query or query == "" then return nil end
    local q = query:lower():gsub("^%s*(.-)%s*$", "%1")
    if q == "" then return nil end

    -- Сначала ищем точное совпадение по нику или DisplayName
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player then
            if plr.Name:lower() == q or plr.DisplayName:lower() == q then
                return plr
            end
        end
    end

    -- Затем ищем по началу ника или DisplayName
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player then
            if plr.Name:lower():sub(1, #q) == q or plr.DisplayName:lower():sub(1, #q) == q then
                return plr
            end
        end
    end
    return nil
end

local _, targetStatusLabel = createParagraph(playersPage, "🎯 Выбранная цель", "Никто не выбран (введите ник ниже)")

createInput(playersPage, "Ник игрока (первые буквы или полный)", "Введите ник...", function(Text)
    targetPlayerInput = Text
    local found = findExactOrPrefixPlayer(targetPlayerInput)
    if found then
        targetStatusLabel.Text = "Цель найдена: " .. found.DisplayName .. " (@" .. found.Name .. ")"
    else
        targetStatusLabel.Text = (Text == "") and "Никто не выбран" or ("Игрок '" .. Text .. "' не найден")
    end
end)

createButton(playersPage, "🌀 Телепортироваться к игроку", function()
    local target = findExactOrPrefixPlayer(targetPlayerInput)
    if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
        local targetHrp = target.Character.HumanoidRootPart
        local destCFrame = targetHrp.CFrame * CFrame.new(0, 0, 3)

        if phantomRunning and ghostSimulationCube then
            ghostSimulationCube.CFrame = destCFrame
        else
            local char = Player.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                char.HumanoidRootPart.CFrame = destCFrame
            end
        end
    end
end)

createToggle(playersPage, "👁 Наблюдать за игроком (Spectate)", function(Value)
    if Value then
        local target = findExactOrPrefixPlayer(targetPlayerInput)
        if target and target.Character and target.Character:FindFirstChildOfClass("Humanoid") then
            spectatingPlayer = target
            workspace.CurrentCamera.CameraSubject = target.Character:FindFirstChildOfClass("Humanoid")
        end
    else
        spectatingPlayer = nil
        if phantomRunning and ghostVisualClone then
            local camTarget = ghostVisualClone:FindFirstChild("Head") or ghostVisualClone.PrimaryPart or ghostSimulationCube
            workspace.CurrentCamera.CameraSubject = camTarget
        elseif Player.Character and Player.Character:FindFirstChildOfClass("Humanoid") then
            workspace.CurrentCamera.CameraSubject = Player.Character:FindFirstChildOfClass("Humanoid")
        end
    end
end)

createToggle(playersPage, "🧲 Быть сзади выбранного игрока", function(Value)
    followBehindEnabled = Value
    if followBehindEnabled then
        local lockedTarget = findExactOrPrefixPlayer(targetPlayerInput)
        if not lockedTarget then
            followBehindEnabled = false
            return
        end

        task.spawn(function()
            while followBehindEnabled and not _G.ScriptDestroyed do
                if not _G.ScriptPaused then
                    -- Если в поле ввели новый ник, обновляем цель; иначе держимся строго за выбранным игроком
                    local currentQueryTarget = findExactOrPrefixPlayer(targetPlayerInput)
                    if currentQueryTarget then
                        lockedTarget = currentQueryTarget
                    end

                    if not lockedTarget or not lockedTarget.Parent then
                        followBehindEnabled = false
                        break
                    end

                    local tChar = lockedTarget.Character
                    local tHrp = tChar and tChar:FindFirstChild("HumanoidRootPart")
                    if tHrp then
                        local behindCFrame = tHrp.CFrame * CFrame.new(0, 0, 3.2)
                        if phantomRunning and ghostSimulationCube then
                            ghostSimulationCube.CFrame = behindCFrame
                        else
                            local myChar = Player.Character
                            local myHrp = myChar and myChar:FindFirstChild("HumanoidRootPart")
                            if myHrp then
                                myHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                myHrp.CFrame = behindCFrame
                            end
                        end
                    end
                end
                RunService.RenderStepped:Wait()
            end
        end)
    end
end)

end

local function findNearestEnemyToCenter()
    local cam = workspace.CurrentCamera
    local center = Vector2.new(cam.ViewportSize.X * 0.5, cam.ViewportSize.Y * 0.5)
    local bestPlr = nil
    local bestScore = math.huge

    local myChar = Player.Character
    local myRoot = (phantomRunning and ghostSimulationCube) or (myChar and myChar:FindFirstChild("HumanoidRootPart"))

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= Player and plr.Character then
            local tHrp = plr.Character:FindFirstChild("HumanoidRootPart")
            local tHum = plr.Character:FindFirstChildOfClass("Humanoid")
            if tHrp and tHum and tHum.Health > 0 then
                local worldDist = myRoot and (myRoot.Position - tHrp.Position).Magnitude or 0
                if worldDist <= 250 then
                    local screenPos, onScreen = cam:WorldToViewportPoint(tHrp.Position)
                    local screenDist = onScreen and (Vector2.new(screenPos.X, screenPos.Y) - center).Magnitude or 9999
                    local score = (worldDist <= 20) and worldDist or (screenDist + worldDist * 0.5)
                    if score < bestScore then
                        bestScore = score
                        bestPlr = plr
                    end
                end
            end
        end
    end
    return bestPlr
end

-- ШИФТЛОК: пока он включён, "мышь" для игры = центр экрана (как заблокированная мышь на ПК)
-- Работает через hookmetamethod; без него шифтлок просто остаётся камерой/поворотом как раньше.
task.spawn(function()
    repeat task.wait(0.5) until shiftLockActive or _G.ScriptDestroyed
    if _G.ScriptDestroyed then return end
    if not (hookmetamethod and checkcaller and getnamecallmethod) then
        warn("[Шифтлок] Экзекутор не поддерживает hookmetamethod: выстрелы в центр недоступны")
        return
    end

    local GuiService = game:GetService("GuiService")
    local UIS = game:GetService("UserInputService")
    local mouse = Player:GetMouse()
    local wrap = newcclosure or function(f) return f end

    local function centerPoint()
        local vs = workspace.CurrentCamera.ViewportSize
        return Vector2.new(vs.X * 0.5, vs.Y * 0.5)
    end

    local function centerRay()
        local c = centerPoint()
        return workspace.CurrentCamera:ViewportPointToRay(c.X, c.Y)
    end

    local function centerRaycast()
        local ray = centerRay()
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        local ignore = {}
        if Player.Character then table.insert(ignore, Player.Character) end
        if ghostVisualClone then table.insert(ignore, ghostVisualClone) end
        if ghostSimulationCube then table.insert(ignore, ghostSimulationCube) end
        params.FilterDescendantsInstances = ignore
        local result = workspace:Raycast(ray.Origin, ray.Direction * 1000, params)
        return ray, result
    end

    local function redirectActive()
        return shiftLockActive and not _G.ScriptPaused and not _G.ScriptDestroyed
    end

    local oldIndex
    oldIndex = hookmetamethod(game, "__index", wrap(function(self, key)
        if self == mouse and not checkcaller() and redirectActive() then
            if key == "Hit" then
                local ray, result = centerRaycast()
                return CFrame.new(result and result.Position or (ray.Origin + ray.Direction * 1000))
            elseif key == "Target" then
                local _, result = centerRaycast()
                return result and result.Instance or nil
            elseif key == "X" then
                return centerPoint().X - GuiService:GetGuiInset().X
            elseif key == "Y" then
                return centerPoint().Y - GuiService:GetGuiInset().Y
            elseif key == "UnitRay" then
                return centerRay()
            end
        end
        return oldIndex(self, key)
    end))

    local oldNamecall
    oldNamecall = hookmetamethod(game, "__namecall", wrap(function(self, ...)
        if self == UIS and not checkcaller() and redirectActive() and getnamecallmethod() == "GetMouseLocation" then
            return centerPoint()
        end
        return oldNamecall(self, ...)
    end))
end)

end

local playerCooldownData = {}

local function registerPlayerTool(pData, toolObj)
    if not toolObj or not toolObj:IsA("Tool") then return end
    local toolName = toolObj.Name
    local slotIdx = pData.NameToSlot[toolName]

    if not slotIdx then
        if #pData.SlotNames < 9 then
            table.insert(pData.SlotNames, toolName)
            slotIdx = #pData.SlotNames
            pData.NameToSlot[toolName] = slotIdx
        else
            slotIdx = 9
        end
    end

    if not pData.HookedTools[toolObj] then
        pData.HookedTools[toolObj] = true
        toolObj.Activated:Connect(function()
            local cd = toolObj:GetAttribute("Cooldown") or toolObj:GetAttribute("CD") or 4
            if type(cd) ~= "number" or cd <= 0 then cd = 4 end
            pData.Cooldowns[slotIdx] = tick() + cd
        end)
    end
end

local function setupPlayerCdTracker(plr, char, hum)
    local pData = {
        Char = char,
        SlotNames = {},
        NameToSlot = {},
        Cooldowns = {},
        HookedTools = {},
        LastAnimTime = 0
    }
    playerCooldownData[plr] = pData

    char.ChildAdded:Connect(function(child)
        if child:IsA("Tool") then registerPlayerTool(pData, child) end
    end)

    local animator = hum:FindFirstChildOfClass("Animator")
    if not animator then return pData end

    animator.AnimationPlayed:Connect(function(track)
        if tick() - pData.LastAnimTime < 0.45 then return end
        local animName = track.Animation and track.Animation.Name:lower() or ""
        if animName:find("walk") or animName:find("run") or animName:find("idle") or animName:find("jump") or animName:find("fall") then
            return
        end
        pData.LastAnimTime = tick()

        local heldTool = char:FindFirstChildOfClass("Tool")
        if heldTool then
            registerPlayerTool(pData, heldTool)
            local sIdx = pData.NameToSlot[heldTool.Name]
            if sIdx and (not pData.Cooldowns[sIdx] or pData.Cooldowns[sIdx] < tick()) then
                local cd = heldTool:GetAttribute("Cooldown") or heldTool:GetAttribute("CD") or 4
                if type(cd) ~= "number" or cd <= 0 then cd = 4 end
                pData.Cooldowns[sIdx] = tick() + cd
            end
            return
        end

        if #pData.SlotNames == 0 then
            for k = 1, 4 do
                local sName = "Скилл " .. k
                table.insert(pData.SlotNames, sName)
                pData.NameToSlot[sName] = k
            end
        end
        for k = 1, #pData.SlotNames do
            if not pData.Cooldowns[k] or pData.Cooldowns[k] < tick() then
                pData.Cooldowns[k] = tick() + 5
                break
            end
        end
    end)

    return pData
end

local function evaluateMathExpression(rawText)
    if not rawText or type(rawText) ~= "string" then return nil, nil end

    local cleaned = rawText:gsub("×", "*"):gsub("х", "*"):gsub("Х", "*"):gsub("x", "*"):gsub("X", "*")
    cleaned = cleaned:gsub("÷", "/"):gsub(":", "/")

    local aStr, op, bStr = cleaned:match("(%-?%d+%.?%d*)%s*([%+%-/%*])%s*(%-?%d+%.?%d*)")
    if aStr and op and bStr then
        local a = tonumber(aStr)
        local b = tonumber(bStr)
        if a and b then
            local res = nil
            if op == "+" then res = a + b
            elseif op == "-" then res = a - b
            elseif op == "*" then res = a * b
            elseif op == "/" and b ~= 0 then res = a / b end

            if res ~= nil then
                local formattedRes = (res % 1 == 0) and tostring(math.floor(res)) or string.format("%.2f", res)
                return string.format("%s %s %s", aStr, op, bStr), formattedRes
            end
        end
    end

    local safeExpr = cleaned:gsub("[^%d%+%-%*/%.%(%)%s]", "")
    if safeExpr:find("[%+%-%*/]") and safeExpr:find("%d") then
        local fn = loadstring("return " .. safeExpr)
        if fn then
            local ok, val = pcall(fn)
            if ok and type(val) == "number" and val == val and math.abs(val) < 1e15 then
                local formattedRes = (val % 1 == 0) and tostring(math.floor(val)) or string.format("%.2f", val)
                return safeExpr, formattedRes
            end
        end
    end

    return nil, nil
end

                                        if not adorneePart:FindFirstChild("PromptWorldMathHint") then

                                        end

                                end
                            end
                        end
                    end
                end

local function stopBendyDive(teleportBack)
    if bendyConnStepped then bendyConnStepped:Disconnect() bendyConnStepped = nil end
    if bendyConnHeartbeat then bendyConnHeartbeat:Disconnect() bendyConnHeartbeat = nil end

    local char = Player.Character
    local realHrp = char and char:FindFirstChild("HumanoidRootPart")
    local realHum = char and char:FindFirstChildOfClass("Humanoid")

    if bendyProxyRig then
        local pHrp = bendyProxyRig:FindFirstChild("HumanoidRootPart")
        if teleportBack and realHrp and pHrp then
            realHrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            realHrp.CFrame = pHrp.CFrame + Vector3.new(0, 1.5, 0)
        end
        pcall(function() bendyProxyRig:Destroy() end)
        bendyProxyRig = nil
    end

    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = (part.Name == "HumanoidRootPart" or part.Name == "Torso" or part.Name == "UpperTorso" or part.Name == "LowerTorso")
            end
        end
    end

    if realHum then
        realHum.PlatformStand = false
        realHum:ChangeState(Enum.HumanoidStateType.GettingUp)
        workspace.CurrentCamera.CameraSubject = realHum
    end
end

local function startBendyDive(hudBtn)
    local char = Player.Character
    local realHrp = char and char:FindFirstChild("HumanoidRootPart")
    local realHum = char and char:FindFirstChildOfClass("Humanoid")
    if not realHrp or not realHum then return end

    local walkAnim, jumpAnim
    local origAnim = char:FindFirstChild("Animate")
    if origAnim then
        local r = origAnim:FindFirstChild("run") or origAnim:FindFirstChild("walk")
        if r and r:FindFirstChildOfClass("Animation") then
            walkAnim = r:FindFirstChildOfClass("Animation"):Clone()
        end
        local j = origAnim:FindFirstChild("jump")
        if j and j:FindFirstChildOfClass("Animation") then
            jumpAnim = j:FindFirstChildOfClass("Animation"):Clone()
        end
    end

    char.Archivable = true
    local proxy = char:Clone()
    char.Archivable = false
    proxy.Name = "BendySurfacePuppet"

    for _, obj in ipairs(proxy:GetDescendants()) do
        if obj:IsA("LocalScript") or obj:IsA("Script") or obj:IsA("ForceField") then
            obj:Destroy()
        elseif obj:IsA("BasePart") then
            obj.Anchored = false
            obj.Material = Enum.Material.ForceField
            obj.Color = Color3.fromRGB(35, 35, 45)
            obj.Transparency = (obj.Name == "HumanoidRootPart") and 1 or 0.55
            obj.CanCollide = (obj.Name == "HumanoidRootPart" or obj.Name == "Torso" or obj.Name == "UpperTorso" or obj.Name == "LowerTorso")
        elseif obj:IsA("Decal") or obj:IsA("Texture") then
            obj.Transparency = 0.55
        end
    end

    local pHrp = proxy:FindFirstChild("HumanoidRootPart")
    local pHum = proxy:FindFirstChildOfClass("Humanoid")
    if not pHrp or not pHum then
        proxy:Destroy()
        return
    end

    pHum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
    pHum.WalkSpeed = realHum.WalkSpeed
    pHum.JumpPower = realHum.JumpPower
    pHum.UseJumpPower = realHum.UseJumpPower
    pHum.JumpHeight = realHum.JumpHeight
    pHum.AutoRotate = true
    proxy.Parent = workspace
    bendyProxyRig = proxy

    proxy:PivotTo(realHrp.CFrame)
    workspace.CurrentCamera.CameraSubject = pHum

    local animator = pHum:FindFirstChildOfClass("Animator") or Instance.new("Animator", pHum)
    local walkTrack, jumpTrack
    if walkAnim then
        walkAnim.Parent = proxy
        walkTrack = animator:LoadAnimation(walkAnim)
        walkTrack.Looped = true
    end
    if jumpAnim then
        jumpAnim.Parent = proxy
        jumpTrack = animator:LoadAnimation(jumpAnim)
        jumpTrack.Looped = false
    end

    pHum.Running:Connect(function(spd)
        if not walkTrack then return end
        if spd > 1 then
            if not walkTrack.IsPlaying then walkTrack:Play(0.15) end
            walkTrack:AdjustSpeed(spd / 16)
        else
            if walkTrack.IsPlaying then walkTrack:Stop(0.15) end
        end
    end)

    pHum.StateChanged:Connect(function(_, state)
        if not jumpTrack then return end
        if state == Enum.HumanoidStateType.Jumping or state == Enum.HumanoidStateType.Freefall then
            if not jumpTrack.IsPlaying then jumpTrack:Play(0.1) end
        else
            if jumpTrack.IsPlaying then jumpTrack:Stop(0.15) end
        end
    end)

    bendyConnStepped = RunService.Stepped:Connect(function()
        if not bendyDiveActive or not char.Parent then return end
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                part.CanCollide = false
            end
        end
    end)

    local bob = 0
    bendyConnHeartbeat = RunService.Heartbeat:Connect(function()
        if not bendyDiveActive then return end
        if _G.ScriptDestroyed or not char.Parent or not realHrp.Parent or not proxy.Parent or realHum.Health <= 0 then
            bendyDiveActive = false
            if hudBtn then
                hudBtn.Text = "OFF"
                hudBtn.BackgroundColor3 = Color3.fromRGB(205, 55, 45)
            end
            stopBendyDive(false)
            return
        end

        pHum.WalkSpeed = realHum.WalkSpeed
        pHum:Move(realHum.MoveDirection, false)
        if realHum.Jump then
            pHum.Jump = true
        end

        bob = (bob == 0) and 0.1 or 0
        local voidLimit = workspace.FallenPartsDestroyHeight + 3
        local deepY = math.max(pHrp.Position.Y - 35 + bob, voidLimit)

        realHum:ChangeState(Enum.HumanoidStateType.Physics)
        realHrp.AssemblyLinearVelocity = Vector3.new(0, 0.4, 0)
        realHrp.AssemblyAngularVelocity = Vector3.new(0, 0, 0)
        realHrp.CFrame = CFrame.new(pHrp.Position.X, deepY, pHrp.Position.Z) * (pHrp.CFrame - pHrp.CFrame.Position)
    end)
end

local function destroyBendyHud()
    bendyDiveActive = false
    stopBendyDive(true)
    if bendyHudGui then
        pcall(function() bendyHudGui:Destroy() end)
        bendyHudGui = nil
    end
end

local function finalizeNewBox()
    if editingBoxData or not draftPointA or not draftPointB then return end
    local boxName = nameBox.Text:gsub("^%s*(.-)%s*$", "%1")
    if boxName == "" then return end
    if draftMarkerA then draftMarkerA:Destroy() draftMarkerA = nil end
    if draftMarkerB then draftMarkerB:Destroy() draftMarkerB = nil end
    local mA = createZoneMarker(draftPointA, boxName .. " [А]", Color3.fromRGB(255, 220, 60))
    local mB = createZoneMarker(draftPointB, boxName .. " [Б]", Color3.fromRGB(60, 220, 255))
    local vis = buildBoxVisual(draftPointA, draftPointB, boxName, "")
    table.insert(_G.SavedZoneBoxes, {Name = boxName, PointA = draftPointA, PointB = draftPointB, MarkerA = mA, MarkerB = mB, Visual = vis, Rule = "", Active = true})
    draftPointA, draftPointB, nameBox.Text = nil, nil, ""
    updateCreatorUI()
    refreshBoxesList()
end

                    local char = Player.Character

                    if hrp then
                        for _, bData in ipairs(_G.SavedZoneBoxes) do
                            if bData.Active and bData.PointA and bData.PointB and (bData.Rule == "+" or bData.Rule == "-") then
                                local minX = math.min(bData.PointA.X, bData.PointB.X) - 5
                                local maxX = math.max(bData.PointA.X, bData.PointB.X) + 5
                                local midY = (bData.PointA.Y + bData.PointB.Y) / 2
                                local minY, maxY = midY - 10, midY + 10
                                local minZ = math.min(bData.PointA.Z, bData.PointB.Z) - 5
                                local maxZ = math.max(bData.PointA.Z, bData.PointB.Z) + 5
                                local pos = hrp.Position
                                local isInside = (pos.X >= minX and pos.X <= maxX) and (pos.Y >= minY and pos.Y <= maxY) and (pos.Z >= minZ and pos.Z <= maxZ)

                                if bData.Rule == "+" and not isInside then
                                    local clamped = Vector3.new(math.clamp(pos.X, minX + 1.5, maxX - 1.5), math.clamp(pos.Y, minY + 1.5, maxY - 1.5), math.clamp(pos.Z, minZ + 1.5, maxZ - 1.5))
                                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                    hrp.CFrame = CFrame.new(clamped) * (hrp.CFrame - hrp.CFrame.Position)
                                elseif bData.Rule == "-" and isInside then
                                    local dL, dR, dT, dB = math.abs(pos.X - minX), math.abs(maxX - pos.X), math.abs(pos.Z - minZ), math.abs(maxZ - pos.Z)
                                    local mD = math.min(dL, dR, dT, dB)
                                    local pushPos = (mD == dL and Vector3.new(minX - 6, pos.Y, pos.Z)) or (mD == dR and Vector3.new(maxX + 6, pos.Y, pos.Z)) or (mD == dT and Vector3.new(pos.X, pos.Y, minZ - 6)) or Vector3.new(pos.X, pos.Y, maxZ + 6)
                                    hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                                    hrp.CFrame = CFrame.new(pushPos) * (hrp.CFrame - hrp.CFrame.Position)
                                end
                            end
                        end
                    end
                end
                RunService.Heartbeat:Wait()
            end
        end)
    end
end)

local function recalculateOrders()
    orderCounter = 0
    local sorted = {}
    for _, pt in ipairs(savedPoints) do
        if pt.Order and pt.Order > 0 then table.insert(sorted, pt) end

    table.sort(sorted, function(a,b) return a.Order < b.Order end)
    for i, pt in ipairs(sorted) do
        pt.Order = i
        if pt.NumBox then pt.NumBox.Text = tostring(i) end
        orderCounter = i
    end
end

local function moveToDest(hrp, dest, abortFn)
    if isTeleportMode then
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.CFrame = CFrame.new(dest)
        return
    end
    local wasAnc = hrp.Anchored
    if not dropOnPoint then hrp.Anchored = true end
    while hrp and hrp.Parent and not _G.ScriptDestroyed do
        if abortFn and abortFn() then break end
        local diff = dest - hrp.Position
        local dist = diff.Magnitude
        if dist <= 1.8 then
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
            hrp.CFrame = CFrame.new(dest)
            break
        end
        local dt = RunService.Heartbeat:Wait()
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        hrp.CFrame = CFrame.new(hrp.Position + diff.Unit * math.min(dist, ziplineSpeedValue * dt))
    end
    if hrp and hrp.Parent and not invisActive then hrp.Anchored = wasAnc end
end

local ziplineActive = false
runZiplineBtn.MouseButton1Click:Connect(function()
    if ziplineActive then ziplineActive = false return end
    local char = Player.Character
    local hrp = (invisActive and ghostSimulationCube) or (godmodeEnabled and godGroundCube) or (char and char:FindFirstChild("HumanoidRootPart"))
    if not hrp then return end

    ziplineActive = true
    runZiplineBtn.BackgroundColor3 = Color3.fromRGB(120, 100, 20)
    runZiplineBtn.Text = "🛑 Стоп"

    task.spawn(function()
        repeat
            local ordered = {}
            for _, pt in ipairs(savedPoints) do
                if pt.Order and pt.Order > 0 then table.insert(ordered, pt) end
            end
            table.sort(ordered, function(a, b) return a.Order < b.Order end)
            if #ordered == 0 then break end

            for _, pt in ipairs(ordered) do
                if not ziplineActive or _G.ScriptDestroyed then break end
                local r = (invisActive and ghostSimulationCube) or (godmodeEnabled and godGroundCube) or (Player.Character and Player.Character:FindFirstChild("HumanoidRootPart"))
                if not r then break end
                moveToDest(r, pt.Pos + Vector3.new(0, 3, 0), function() return not ziplineActive end)
                if ziplineDelayValue > 0 and ziplineActive then task.wait(ziplineDelayValue) end
            end
            task.wait(0.05)
        until not isZiplineLoop or not ziplineActive or _G.ScriptDestroyed

        ziplineActive = false
        runZiplineBtn.BackgroundColor3 = Color3.fromRGB(220, 180, 40)
        runZiplineBtn.Text = "🚀 Старт"
    end)
end)

local function bindDeathTracker(char)
    if not char then return end
    local hum = char:WaitForChild("Humanoid", 5)
    local hrp = char:WaitForChild("HumanoidRootPart", 5)
    if hum and hrp then
        hum.Died:Connect(function() if hrp and hrp.Parent then lastDeathPos = hrp.Position end end)
    end
end
if Player.Character then task.spawn(bindDeathTracker, Player.Character) end

Player.CharacterAdded:Connect(function(newChar)
    task.spawn(bindDeathTracker, newChar)
    if not deathZiplineEnabled or _G.ScriptPaused or _G.ScriptDestroyed then return end
    local newHrp = newChar:WaitForChild("HumanoidRootPart", 5)
    if not newHrp then return end
    task.wait(0.45)

    local ordered = {}
    for _, pt in ipairs(savedPoints) do
        if pt.Order and pt.Order > 0 then table.insert(ordered, pt) end
    end
    table.sort(ordered, function(a, b) return a.Order < b.Order end)
    if #ordered == 0 then return end

    local backPos = lastDeathPos or newHrp.Position
    for _, pt in ipairs(ordered) do
        if not deathZiplineEnabled or _G.ScriptDestroyed or not newHrp.Parent then break end
        moveToDest(newHrp, pt.Pos + Vector3.new(0, 3, 0), function() return not deathZiplineEnabled end)
        if ziplineDelayValue > 0 then task.wait(ziplineDelayValue) end
    end
    if deathZiplineEnabled and newHrp.Parent and backPos then
        moveToDest(newHrp, backPos + Vector3.new(0, 3, 0), function() return not deathZiplineEnabled end)
    end
end)
